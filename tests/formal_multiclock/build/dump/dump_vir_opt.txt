; VIR: portable LLVM IR using the VIR runtime ABI
@"destination_pulse" = global i64 zeroinitializer
@"rst_n" = global i1 zeroinitializer
@"source_epoch" = global i64 zeroinitializer
@"source_request" = global i1 zeroinitializer
@"synchronized_request" = global i64 zeroinitializer
@"synchronizer_stage" = global i64 zeroinitializer

define void @"multiclock_bridge.combinational"() {
entry:
  ret void
}

define void @"multiclock_bridge.clocked"() {
entry:
  %0 = load i1, ptr @"rst_n"
  %1 = xor i1 %0, true
  br i1 %1, label %if.then.0, label %if.else.1
if.then.0:
  %2 = zext i1 0 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"source_epoch", i64 %2, i32 1)
  br label %if.end.2
if.else.1:
  %3 = load i1, ptr @"source_request"
  br i1 %3, label %if.then.3, label %if.else.4
if.end.2:
  %4 = load i1, ptr @"rst_n"
  %5 = xor i1 %4, true
  br i1 %5, label %if.then.6, label %if.else.7
if.then.3:
  %6 = load i1, ptr @"source_epoch"
  %7 = xor i1 %6, true
  %8 = zext i1 %7 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"source_epoch", i64 %8, i32 1)
  br label %if.end.5
if.else.4:
  br label %if.end.5
if.end.5:
  br label %if.end.2
if.then.6:
  %9 = zext i1 0 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"synchronizer_stage", i64 %9, i32 1)
  %10 = zext i1 0 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"synchronized_request", i64 %10, i32 1)
  %11 = zext i1 0 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"destination_pulse", i64 %11, i32 1)
  br label %if.end.8
if.else.7:
  %12 = load i1, ptr @"source_request"
  %13 = zext i1 %12 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"synchronizer_stage", i64 %13, i32 1)
  %14 = load i1, ptr @"synchronizer_stage"
  %15 = zext i1 %14 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"synchronized_request", i64 %15, i32 1)
  %16 = load i1, ptr @"synchronizer_stage"
  %17 = load i1, ptr @"synchronized_request"
  %18 = xor i1 %17, true
  %19 = and i1 %16, %18
  %20 = zext i1 %19 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"destination_pulse", i64 %20, i32 1)
  br label %if.end.8
if.end.8:
  ret void
}
