; VIR: portable LLVM IR using the VIR runtime ABI
@"a" = global i8 zeroinitializer
@"b" = global i8 zeroinitializer
@"carry" = global i1 zeroinitializer
@"clk" = global i1 zeroinitializer
@"sum" = global i8 zeroinitializer
@".instance.dut" = private unnamed_addr constant [4 x i8] c"dut\00"
@".io.a" = private unnamed_addr constant [2 x i8] c"a\00"
@".io.b" = private unnamed_addr constant [2 x i8] c"b\00"
@".io.carry" = private unnamed_addr constant [6 x i8] c"carry\00"
@".io.clk" = private unnamed_addr constant [4 x i8] c"clk\00"
@".io.sum" = private unnamed_addr constant [4 x i8] c"sum\00"

define void @"adder_tb.combinational"() {
entry:
  store i8 3, ptr @"b"
  ret void
}

define void @"adder_tb.clocked"() {
entry:
  ret void
}

define void @"adder_tb.initial"() {
entry:
  %0 = trunc i32 0 to i1
  store i1 %0, ptr @"clk"
  %1 = trunc i32 5 to i8
  store i8 %1, ptr @"a"
  call void @"vir.runtime.time.delay"(i32 1)
  %2 = trunc i32 1 to i1
  store i1 %2, ptr @"clk"
  call void @"vir.runtime.time.delay"(i32 1)
  %3 = load i1, ptr @"carry"
  %4 = load i8, ptr @"sum"
  %concat.0.high = zext i1 %3 to i9
  %concat.0.shift = shl i9 %concat.0.high, 8
  %concat.0.low = zext i8 %4 to i9
  %5 = or i9 %concat.0.shift, %concat.0.low
  %6 = icmp eq i9 %5, 8
  call void @"vir.runtime.assert.check"(i1 %6)
  call void @"vir.runtime.finish"()
  ret void
}

declare void @"vir.runtime.time.delay"(i32)

declare void @"vir.runtime.assert.check"(i1)

define void @"adder_tb.transport"() {
entry:
  %7 = call i1 @"vir.runtime.io.read.i1"(i32 0, ptr @".io.clk")
  %8 = call i32 @"vir.runtime.transport.export.i1"(ptr @"clk", i1 %7)
  %9 = call i8 @"vir.runtime.io.read.i8"(i32 0, ptr @".io.a")
  %10 = call i32 @"vir.runtime.transport.export.i8"(ptr @"a", i8 %9)
  %11 = call i8 @"vir.runtime.io.read.i8"(i32 0, ptr @".io.b")
  %12 = call i32 @"vir.runtime.transport.export.i8"(ptr @"b", i8 %11)
  %evaluate.1 = call {i32, ptr} @"vir.runtime.transport.schedule"(ptr @".instance.dut")
  %13 = extractvalue {i32, ptr} %evaluate.1, 0
  %14 = extractvalue {i32, ptr} %evaluate.1, 1
  %import.2 = call i64 @"vir.runtime.transport.import.i8"(ptr @"sum", i1 false)
  %import.2.status = and i64 %import.2, 255
  %15 = trunc i64 %import.2.status to i32
  %import.2.data = lshr i64 %import.2, 8
  %16 = trunc i64 %import.2.data to i8
  %import.2.xmask = lshr i64 %import.2, 16
  %17 = trunc i64 %import.2.xmask to i8
  %import.2.zmask = lshr i64 %import.2, 24
  %18 = trunc i64 %import.2.zmask to i8
  call void @"vir.runtime.io.write.i8"(i32 1, ptr @".io.sum", i8 %16, i8 %17, i8 %18)
  %import.3 = call i64 @"vir.runtime.transport.import.i1"(ptr @"carry", i1 false)
  %import.3.status = and i64 %import.3, 255
  %19 = trunc i64 %import.3.status to i32
  %import.3.data = lshr i64 %import.3, 8
  %20 = trunc i64 %import.3.data to i1
  %import.3.xmask = lshr i64 %import.3, 16
  %21 = trunc i64 %import.3.xmask to i1
  %import.3.zmask = lshr i64 %import.3, 24
  %22 = trunc i64 %import.3.zmask to i1
  call void @"vir.runtime.io.write.i1"(i32 1, ptr @".io.carry", i1 %20, i1 %21, i1 %22)
  ret void
}
