; VIR: portable LLVM IR using the VIR runtime ABI
@"grant" = global i64 zeroinitializer
@"request" = global i1 zeroinitializer
@"rst_n" = global i1 zeroinitializer

define void @"btor2_pipeline.combinational"() {
entry:
  ret void
}

define void @"btor2_pipeline.clocked"() {
entry:
  %0 = load i1, ptr @"rst_n"
  %1 = xor i1 %0, true
  br i1 %1, label %if.then.0, label %if.else.1
if.then.0:
  %2 = zext i1 0 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"grant", i64 %2, i32 1)
  br label %if.end.2
if.else.1:
  %3 = load i1, ptr @"request"
  %4 = zext i1 %3 to i64
  call void @"vir.runtime.nba.store.u64"(ptr @"grant", i64 %4, i32 1)
  br label %if.end.2
if.end.2:
  ret void
}

entry:
  ret void
}
